@app.route('/api/user/login', methods=['POST'])
def user_login():
    data = request.get_json()
    username = data.get('username')
    password = data.get('password')
    if not username or not password:
        return jsonify({'code': 400, 'msg': '账号或密码不能为空'})
    user = User.query.filter_by(username=username).first()
    if not user or user.password != md5(password):
        return jsonify({'code': 401, 'msg': '账号或密码错误'})
    if user.status == 'disabled':
        return jsonify({'code': 403, 'msg': '账号已被禁用'})
    token = generate_token(user.id)
    return jsonify({'code': 200, 'data': {'token': token, 'user': user.to_dict()}})
@app.route('/api/user/register', methods=['POST'])
def user_register():
    data = request.get_json()
    username = data.get('username')
    password = data.get('password')
    phone = data.get('phone')
    if len(username) < 4:
        return jsonify({'code': 400, 'msg': '账号长度不能少于4位'})
    if len(password) < 6:
        return jsonify({'code': 400, 'msg': '密码长度不能少于6位'})
    if User.query.filter_by(username=username).first():
        return jsonify({'code': 400, 'msg': '该账号已被注册'})
    user = User(username=username, password=md5(password), phone=phone)
    db.session.add(user)
    db.session.commit()
    return jsonify({'code': 200, 'data': {'user_id': user.id}})

@app.route('/api/plan/generate', methods=['POST'])
@login_required
def generate_plan():
    user_id = g.user_id
    questionnaire = Questionnaire.query.filter_by(user_id=user_id).order_by(Questionnaire.id.desc()).first()
    if not questionnaire:
        return jsonify({'code': 400, 'msg': '请先完成健康问卷'})
    height = questionnaire.height_cm
    weight = questionnaire.current_weight_kg
    age = questionnaire.age
    gender = questionnaire.gender
    if gender == 'male':
        bmr = 10 * weight + 6.25 * height - 5 * age + 5
    else:
        bmr = 10 * weight + 6.25 * height - 5 * age - 161
    target_weight = questionnaire.target_weight_kg
    goal_cycle = questionnaire.goal_cycle_days
    weight_diff = weight - target_weight
    daily_deficit = (weight_diff * 7700) / goal_cycle
    calorie_target = int(bmr - daily_deficit)
    diet_suggestion = generate_diet_suggestion(questionnaire.diet_preference, calorie_target)
    workout_suggestion = generate_workout_suggestion(questionnaire.sport_base)
    solar_term = get_current_solar_term()
    solar_advice = get_solar_health_advice(solar_term)
    plan = PersonalizedPlan(
        user_id=user_id,
        questionnaire_id=questionnaire.id,
        plan_type='system',
        bmr=bmr,
        daily_calorie_target=calorie_target,
        diet_suggestion=diet_suggestion,
        workout_suggestion=workout_suggestion + '\n【节气养生】' + solar_advice,
        valid_days=goal_cycle
    )
    db.session.add(plan)
    db.session.commit()
    return jsonify({'code': 200, 'data': {'plan_id': plan.id, 'plan': plan.to_dict()}})

@app.route('/api/community/post', methods=['POST'])
@login_required
def create_post():
    data = request.get_json()
    user_id = g.user_id
    title = data.get('title')
    content = data.get('content')
    if not title or not content:
        return jsonify({'code': 400, 'msg': '标题和内容不能为空'})
    if check_sensitive_words(title) or check_sensitive_words(content):
        return jsonify({'code': 400, 'msg': '内容包含违规信息'})
    post = CommunityPost(
        user_id=user_id,
        title=title,
        content=content,
        category=data.get('category', 'share'),
        image_urls=json.dumps(data.get('image_urls', []))
    )
    db.session.add(post)
    db.session.commit()
    award_points(user_id, 'post', 10)
    return jsonify({'code': 200, 'data': {'post_id': post.id}})

# ========== 6. 积分兑换接口 ==========
@app.route('/api/points/exchange', methods=['POST'])
@login_required
def exchange_medal():
    data = request.get_json()
    user_id = g.user_id
    medal_id = data.get('medal_id')
    medal = Medal.query.get(medal_id)
    if not medal or medal.status != 'on_sale':
        return jsonify({'code': 400, 'msg': '勋章不存在或已下架'})
    account = PointAccount.query.filter_by(user_id=user_id).first()
    if account.available_point < medal.required_points:
        return jsonify({'code': 400, 'msg': '积分不足'})
    account.available_point -= medal.required_points
    user_medal = UserMedal(user_id=user_id, medal_id=medal_id)
    db.session.add(user_medal)
    db.session.commit()
    return jsonify({'code': 200, 'data': {'remaining_points': account.available_point}})

@app.route('/api/coach/plan/optimize', methods=['POST'])
@login_required
def optimize_plan():
    data = request.get_json()

    user_id = data.get('user_id')
    original_plan_id = data.get('original_plan_id')

        return jsonify({'code': 400, 'msg': '原方案不存在'})
    new_plan = PersonalizedPlan(
        user_id=user_id,
        questionnaire_id=original.questionnaire_id,
        plan_type='coach_optimized',
        coach_id=coach_id,
        bmr=original.bmr,
        daily_calorie_target=original.daily_calorie_target,
        diet_suggestion=data.get('diet_suggestion'),
        workout_suggestion=data.get('workout_suggestion'),
        valid_days=data.get('valid_days')
    )
    db.session.add(new_plan)
    original.status = 'optimized'
    db.session.commit()
    return jsonify({'code': 200, 'data': {'new_plan_id': new_plan.id}})

@app.route('/api/coach/consult/reply', methods=['POST'])
@login_required
def reply_consult():
    data = request.get_json()
    coach_id = g.user_id
    consult_id = data.get('consult_id')
    consult = OnlineConsult.query.get(consult_id)
    if not consult or consult.coach_user_id != coach_id:
        return jsonify({'code': 403, 'msg': '无权操作'})
    if consult.status != 'pending':
        return jsonify({'code': 400, 'msg': '该咨询已回复'})
    consult.reply = data.get('reply')
    consult.status = 'replied'
    consult.reply_time = datetime.now()
    db.session.commit()
    return jsonify({'code': 200})

# ========== 10. 管理员审核教练接口 ==========
@app.route('/api/admin/coach/audit', methods=['POST'])
@admin_required
def audit_coach():
    data = request.get_json()
    cert_id = data.get('cert_id')
    result = data.get('result')
    reason = data.get('reason')
    cert = CoachCertification.query.get(cert_id)
    if not cert:
        return jsonify({'code': 404, 'msg': '申请不存在'})
    cert.status = result
    cert.reject_reason = reason if result == 'rejected' else None
    if result == 'approved':
        user = User.query.get(cert.user_id)
        user.role = 'coach'
    db.session.commit()
    return jsonify({'code': 200})

# ========== 11. uni-app 登录页面核心代码 ==========
<template>
<view class="login-container">
<input v-model="username" placeholder="请输入账号" />
<input v-model="password" type="password" placeholder="请输入密码" />
<button @click="handleLogin">登录</button>
<navigator url="/pages/register/index">去注册</navigator>
</view>
</template>
<script>
export default {
data() {
return { username: '', password: '' }
},
methods: {
async handleLogin() {
if (!this.username || !this.password) {
uni.showToast({ title: '账号密码不能为空', icon: 'none' })
return
}
const res = await uni.request({
url: 'https://api.fitpal.com/api/user/login',
method: 'POST',
data: { username: this.username, password: this.password }
})
if (res.data.code === 200) {
uni.setStorageSync('token', res.data.data.token)
uni.switchTab({ url: '/pages/home/index' })
} else {
uni.showToast({ title: res.data.msg, icon: 'none' })
}
}
}
}
</script>

send_system_message(content.user_id, f'您的内容因"{reason}"被{result}', content_type)
db.session.commit()
return jsonify({'code': 200, 'data': {'content_id': content_id, 'status': result}})
